Private Sub UserForm_Initialize()
    Dim v As Variant
    On Error GoTo Echec
    m_bChargement = True

    Me.Caption = "Horizon_Contrat - saisie d'un relevé journalier"
    Me.Width = 522
    Me.Height = 512
    ConstruireControles

    v = ListeMagasinsOuverts()
    If Not IsEmpty(v) Then cboMagasin.List = v
    If cboMagasin.ListCount > 0 Then cboMagasin.ListIndex = 0
    txtDate.Text = Format$(DatePreremplie(), "dd/mm/yyyy")
    optSurPlace.Value = True

    m_bChargement = False
    ActualiserMagasin
    ChoisirCanal CanalSuivantLibre(CodeMagasin(), DateOuAujourdhui(), 0)
    Exit Sub
Echec:
    m_bChargement = False
    MsgBox "Initialisation du formulaire : " & Err.Description, vbExclamation, "Horizon_Contrat"
End Sub

' Exit avec Cancel : refuse de quitter le champ tant que la date est invalide.
' Le contrôle de doublon n'est PAS fait ici : il dépend du canal, choisi après
' la date dans l'ordre de tabulation ; il est affiché en direct et bloquant
' à la validation.
Private Sub txtDate_Exit(ByVal Cancel As MSForms.ReturnBoolean)
    Dim d As Date, sMsg As String
    If m_bChargement Then Exit Sub
    If Trim$(txtDate.Text) = "" Then Exit Sub

    If Not LireDateFr(txtDate.Text, d) Then
        sMsg = "Date invalide. Format attendu : JJ/MM/AAAA."
    Else
        sMsg = MessageDateHorsPeriode(d)
    End If
    If sMsg <> "" Then
        MsgBox sMsg, vbExclamation, "Horizon_Contrat"
        Cancel = True
        Exit Sub
    End If

    m_bChargement = True
    txtDate.Text = Format$(d, "dd/mm/yyyy")           ' 5/9/26 devient 05/09/2026
    m_bChargement = False
    RecalculerControles
End Sub

Private Sub cmdValider_Click()
    Dim sCode As String, d As Date, sCanal As String, iCanal As Long
    Dim dBrut As Double, dRemise As Double, dCout As Double, dNet As Double
    Dim lTickets As Long, lArticles As Long, sAlertes As String

    On Error GoTo Echec
    RecalculerControles
    If Not m_bValide Then
        MsgBox "Relevé refusé :" & IIf(m_sBloquants <> "", m_sBloquants, vbLf & "- montants incomplets") & _
               vbLf & vbLf & "Corrige les lignes en rouge.", vbExclamation, "Horizon_Contrat"
        Exit Sub
    End If

    sCode = CodeMagasin()
    LireDateFr txtDate.Text, d
    iCanal = IndexCanal()
    sCanal = NomCanal(iCanal)
    dBrut = ValeurNum(txtCABrut.Text)
    dRemise = ValeurNum(txtRemises.Text)
    dCout = ValeurNum(txtCoutMatiere.Text)
    lTickets = CLng(ValeurNum(txtTickets.Text))
    lArticles = CLng(ValeurNum(txtArticles.Text))
    dNet = (dBrut - dRemise) / (1 + CDbl(Param("Taux_TVA_Restauration")))
    sAlertes = Replace$(Mid$(m_sAlertes, 4), vbLf & "- ", " | ")

    EnregistrerReleve sCode, d, sCanal, dBrut, dRemise, dNet, dCout, lTickets, lArticles, sAlertes
    Journaliser "Saisie relevé", 1, IIf(sAlertes <> "", 1, 0), 0, _
                sCode & " " & Format$(d, "dd/mm/yyyy") & " " & sCanal & _
                IIf(sAlertes <> "", " - ALERTE : " & sAlertes, "")

    ChargerDerniersReleves
    MettreAJourProgression
    PreparerReleveSuivant sCode, d, iCanal
    lblInfo.Caption = "Relevé " & sCode & " du " & Format$(d, "dd/mm/yyyy") & " (" & sCanal & ") enregistré" & _
                      IIf(sAlertes <> "", " - alerte tracée dans JOURNAL.", ".")
    Exit Sub
Echec:
    MsgBox "Enregistrement impossible : " & Err.Description, vbExclamation, "Horizon_Contrat"
End Sub

'==============================================================================
' Contrôles de cohérence : 3 bloquants, 3 alertes. Seuils lus dans PARAMETRES.
'==============================================================================
Private Sub RecalculerControles()
    Dim sCode As String, sCanal As String, d As Date, bDate As Boolean, sMsg As String
    Dim dBrut As Double, dRemise As Double, dCout As Double, dNet As Double
    Dim lTickets As Long, lArticles As Long
    Dim dMin As Double, dMax As Double, dRatio As Double, dTM As Double, dRef As Double, sSource As String
    Dim dEcart As Double, dTol As Double, dTaux As Double, dSeuil As Double

    On Error GoTo Echec
    sCode = CodeMagasin()
    sCanal = NomCanal(IndexCanal())
    bDate = LireDateFr(txtDate.Text, d)
    dBrut = ValeurNum(txtCABrut.Text)
    dRemise = ValeurNum(txtRemises.Text)
    dCout = ValeurNum(txtCoutMatiere.Text)
    lTickets = CLng(ValeurNum(txtTickets.Text))
    lArticles = CLng(ValeurNum(txtArticles.Text))
    dNet = (dBrut - dRemise) / (1 + CDbl(Param("Taux_TVA_Restauration")))
    lblCANet.Caption = IIf(dBrut > 0, Format$(dNet, "#,##0.00") & " €", "-")

    m_sBloquants = ""
    m_sAlertes = ""
    m_bValide = (sCode <> "" And dBrut > 0 And dCout > 0 And lTickets > 0 And lArticles > 0 And dRemise < dBrut)

    ' 1. Date dans le mois ouvert (bloquant)
    If Not bDate Then
        Verdict 1, True, ETAT_ATTENTE, "Date : à saisir au format JJ/MM/AAAA"
    Else
        sMsg = MessageDateHorsPeriode(d)
        If sMsg = "" Then
            Verdict 1, True, ETAT_OK, "Date du " & Format$(d, "dddd d mmmm yyyy") & ", dans le mois ouvert"
        Else
            Verdict 1, True, ETAT_KO, sMsg
        End If
    End If

    ' 2. Pas de doublon magasin / date / canal (bloquant)
    If sCode = "" Or Not bDate Then
        Verdict 2, True, ETAT_ATTENTE, "Doublon : en attente du magasin et de la date"
    ElseIf ReleveExiste(sCode, d, sCanal) Then
        Verdict 2, True, ETAT_KO, "Un relevé existe déjà pour " & sCode & " le " & Format$(d, "dd/mm/yyyy") & " (" & sCanal & ")"
    Else
        Verdict 2, True, ETAT_OK, "Aucun relevé existant pour ce magasin, cette date et ce canal"
    End If

    ' 3. Articles par ticket plausibles (bloquant)
    dMin = CDbl(Param("Ratio_Articles_Min"))
    dMax = CDbl(Param("Ratio_Articles_Max"))
    If lTickets <= 0 Or lArticles <= 0 Then
        Verdict 3, True, ETAT_ATTENTE, "Articles par ticket : attendus entre " & Format$(dMin, "0.0") & " et " & Format$(dMax, "0.0")
    Else
        dRatio = lArticles / lTickets
        Verdict 3, True, IIf(dRatio >= dMin And dRatio <= dMax, ETAT_OK, ETAT_KO), _
                "Articles par ticket : " & Format$(dRatio, "0.00") & " (plage admise " & Format$(dMin, "0.0") & " - " & Format$(dMax, "0.0") & ")"
    End If

    ' 4. Ticket moyen proche de la référence (alerte)
    dTol = CDbl(Param("Seuil_Ecart_Ticket_Moyen"))
    If sCode = "" Or dNet <= 0 Or lTickets <= 0 Then
        Verdict 4, False, ETAT_ATTENTE, "Ticket moyen : écart toléré " & Format$(dTol, "0%") & " à la moyenne"
    Else
        dTM = dNet / lTickets
        dRef = TicketMoyenReference(sCode, IIf(bDate, d, Date), sSource)
        If dRef <= 0 Then
            Verdict 4, False, ETAT_ATTENTE, "Ticket moyen " & Format$(dTM, "0.00") & " € : pas de référence pour ce magasin"
        Else
            dEcart = dTM / dRef - 1
            Verdict 4, False, IIf(Abs(dEcart) <= dTol, ETAT_OK, ETAT_KO), _
                    "Ticket moyen " & Format$(dTM, "0.00") & " € : " & Format$(dEcart, "+0.0%;-0.0%") & _
                    " vs " & sSource & " (" & Format$(dRef, "0.00") & " €, tolérance " & Format$(dTol, "0%") & ")"
        End If
    End If

    ' 5. Remises sous le plafond (alerte)
    dSeuil = CDbl(Param("Plafond_Remise"))
    If dBrut <= 0 Then
        Verdict 5, False, ETAT_ATTENTE, "Remises : plafond " & Format$(dSeuil, "0%") & " du CA brut"
    Else
        dTaux = dRemise / dBrut
        Verdict 5, False, IIf(dTaux <= dSeuil, ETAT_OK, ETAT_KO), _
                "Remises : " & Format$(dTaux, "0.0%") & " du CA brut (plafond " & Format$(dSeuil, "0%") & ")"
    End If

    ' 6. Coût matière sous le seuil du magasin (alerte)
    If sCode = "" Or dNet <= 0 Or dCout <= 0 Then
        Verdict 6, False, ETAT_ATTENTE, "Coût matière : comparé au taux de coût de l'exercice du magasin"
    Else
        dTaux = dCout / dNet
        dSeuil = TauxCoutMagasin(sCode) + CDbl(Param("Tolerance_Cout_Matiere_Pt"))
        Verdict 6, False, IIf(dTaux <= dSeuil, ETAT_OK, ETAT_KO), _
                "Coût matière : " & Format$(dTaux, "0.0%") & " du CA net HT (seuil du magasin " & Format$(dSeuil, "0.0%") & ")"
    End If
    Exit Sub

Echec:
    m_bValide = False
    lblInfo.Caption = "Contrôles impossibles : " & Err.Description
End Sub

Private Sub Verdict(ByVal i As Long, ByVal bBloquant As Boolean, ByVal iEtat As Long, ByVal sTexte As String)
    Select Case iEtat
        Case ETAT_OK
            m_lblCtrl(i).Caption = "OK          " & sTexte
            m_lblCtrl(i).ForeColor = RGB(11, 110, 95)
        Case ETAT_KO
            If bBloquant Then
                m_lblCtrl(i).Caption = "BLOQUANT    " & sTexte
                m_lblCtrl(i).ForeColor = RGB(180, 40, 40)
                m_sBloquants = m_sBloquants & vbLf & "- " & sTexte
                m_bValide = False
            Else
                m_lblCtrl(i).Caption = "ALERTE      " & sTexte
                m_lblCtrl(i).ForeColor = RGB(190, 110, 0)
                m_sAlertes = m_sAlertes & vbLf & "- " & sTexte
            End If
        Case Else
            m_lblCtrl(i).Caption = "...         " & sTexte
            m_lblCtrl(i).ForeColor = RGB(140, 148, 144)
            If bBloquant Then m_bValide = False
    End Select
End Sub

Private Function MessageDateHorsPeriode(ByVal d As Date) As String
    Dim dMO As Date
    dMO = DebutDeMois(CDate(Param("Mois_Ouvert")))
    If d < dMO Or d > FinDeMois(dMO) Then
        MessageDateHorsPeriode = "La date doit être dans le mois ouvert à la saisie (" & Format$(dMO, "mmmm yyyy") & _
                                 "). Pour changer de mois : paramètre Mois_Ouvert dans PARAMETRES."
    ElseIf d > Date Then
        MessageDateHorsPeriode = "Date dans le futur : un relevé se saisit une fois la journée terminée."
    End If
End Function

'==============================================================================
' Magasin, canal, liste, progression
'==============================================================================
Private Sub ActualiserMagasin()
    Dim sCode As String, bDrive As Boolean
    On Error GoTo Echec
    sCode = CodeMagasin()
    If sCode = "" Then Exit Sub

    bDrive = MagasinADrive(sCode)
    m_bChargement = True
    optDrive.Enabled = bDrive
    If Not bDrive And optDrive.Value = True Then optSurPlace.Value = True
    m_bChargement = False

    lblInfo.Caption = cboMagasin.List(cboMagasin.ListIndex, 1) & "  ·  drive : " & IIf(bDrive, "oui", "non") & _
                      "  ·  taux de coût matière de l'exercice : " & Format$(TauxCoutMagasin(sCode), "0.0%")
    ChargerDerniersReleves
    MettreAJourProgression
    RecalculerControles
    Exit Sub
Echec:
    m_bChargement = False
    lblInfo.Caption = "Lecture du magasin impossible : " & Err.Description
End Sub

Private Function CodeMagasin() As String
    If cboMagasin.ListIndex >= 0 Then CodeMagasin = CStr(cboMagasin.List(cboMagasin.ListIndex, 0))
End Function

Private Function IndexCanal() As Long
    If optEmporter.Value = True Then
        IndexCanal = 2
    ElseIf optLivraison.Value = True Then
        IndexCanal = 3
    ElseIf optDrive.Value = True Then
        IndexCanal = 4
    Else
        IndexCanal = 1
    End If
End Function

Private Sub ChoisirCanal(ByVal i As Long)
    m_bChargement = True
    Select Case i
        Case 1: optSurPlace.Value = True
        Case 2: optEmporter.Value = True
        Case 3: optLivraison.Value = True
        Case 4: If optDrive.Enabled Then optDrive.Value = True
    End Select
    m_bChargement = False
    RecalculerControles
End Sub

Private Function DatePreremplie() As Date
    Dim dMO As Date, d As Date
    dMO = DebutDeMois(CDate(ParamOuDefaut("Mois_Ouvert", Date)))
    d = Date - 1
    If d < dMO Or d > FinDeMois(dMO) Then d = dMO
    DatePreremplie = d
End Function

Private Function DateOuAujourdhui() As Date
    Dim d As Date
    If LireDateFr(txtDate.Text, d) Then
        DateOuAujourdhui = d
    Else
        DateOuAujourdhui = Date
    End If
End Function

' Après une validation : montants vidés, canal suivant non encore saisi pour
' cette date ; si les canaux du jour sont tous saisis, jour suivant.
' Cinq relevés d'affilée se saisissent ainsi au clavier seul.
Private Sub PreparerReleveSuivant(ByVal sCode As String, ByVal d As Date, ByVal iCanal As Long)
    Dim iSuivant As Long, d2 As Date
    ViderMontants
    iSuivant = CanalSuivantLibre(sCode, d, iCanal)
    If iSuivant = 0 Then
        d2 = d + 1
        If MessageDateHorsPeriode(d2) = "" Then
            m_bChargement = True
            txtDate.Text = Format$(d2, "dd/mm/yyyy")
            m_bChargement = False
            iSuivant = CanalSuivantLibre(sCode, d2, 0)
        End If
    End If
    If iSuivant > 0 Then ChoisirCanal iSuivant Else RecalculerControles
    txtCABrut.SetFocus
End Sub

Private Sub ViderMontants()
    m_bChargement = True
    txtCABrut.Text = ""
    txtRemises.Text = ""
    txtCoutMatiere.Text = ""
    txtTickets.Text = ""
    txtArticles.Text = ""
    m_bChargement = False
End Sub

Private Function SaisieEnCours() As Boolean
    SaisieEnCours = (Len(txtCABrut.Text & txtRemises.Text & txtCoutMatiere.Text & txtTickets.Text & txtArticles.Text) > 0)
End Function

Private Sub ChargerDerniersReleves()
    Dim v As Variant
    On Error GoTo Echec
    lstReleves.Clear
    v = DerniersReleves(CodeMagasin(), 5)
    If IsEmpty(v) Then
        lstReleves.AddItem "(aucun relevé saisi pour ce magasin)"
    Else
        lstReleves.List = v
    End If
    Exit Sub
Echec:
    lstReleves.Clear
    lstReleves.AddItem "(lecture impossible : " & Err.Description & ")"
End Sub

' Barre de progression maison : un Label coloré dont on pilote la largeur.
' Pas de contrôle ActiveX ProgressBar : absent de nombreux postes, il casse
' le formulaire à l'ouverture chez les autres.
Private Sub MettreAJourProgression()
    Dim sCode As String, dMO As Date, n As Long, nTotal As Long, dPart As Double
    On Error GoTo Echec
    sCode = CodeMagasin()
    If sCode = "" Then Exit Sub
    dMO = DebutDeMois(CDate(Param("Mois_Ouvert")))
    n = NbRelevesMois(sCode, dMO)
    nTotal = Day(FinDeMois(dMO)) * NbCanauxMagasin(sCode)
    If nTotal > 0 Then dPart = n / nTotal
    If dPart > 1 Then dPart = 1
    lblProgBarre.Width = IIf(dPart * m_dLargeurProg < 1, 1, dPart * m_dLargeurProg)
    lblProgBarre.Visible = (n > 0)
    lblProgTexte.Caption = n & " / " & nTotal
    fraProgression.Caption = "Relevés de " & Format$(dMO, "mmmm yyyy") & " pour ce magasin (jours x canaux)"
    Exit Sub
Echec:
    lblProgTexte.Caption = "?"
End Sub
